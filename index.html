hubspot_mcp_tool.ts
server.registerTool("get_deal_summary", {
  description: "Read-only summary of a HubSpot deal for agent context",
  inputSchema: { dealId: z.string() },

  async handler({ dealId }) {
    // scoped token, read-only – the agent never touches raw creds
    const deal = await hubspot.deals.get(dealId, { readOnly: true });

    return {
      stage: deal.stage,
      amount: deal.amount,
      lastActivity: deal.timeline.latest()
    };
  }
});
05 · Under the hood

How a sync actually survives production

webhook_handler.php
class StripeWebhookHandler
{
    public function handle(Request $req)
    {
        // 1. verify the signature before trusting anything
        if (!Signature::verify($req, config('stripe.secret')))
            abort(400);

        // 2. de-dupe on Stripe's event id (idempotency)
        if (ProcessedEvent::seen($req->event_id)) return response()->noContent();

        // 3. queue it – HubSpot's API waits for no one
        SyncToHubspot::dispatch($req->payload)
            ->onQueue('hubspot-sync')
            ->backoff([5, 30, 120, 600]);

        return response()->noContent();
    }
}
bulk_backfill.js
async function backfillContacts(records) {
  const chunks = chunk(records, 100);

  for (const batch of chunks) {
    const res = await hubspot.batchUpsert(batch);

    // respect HubSpot's rate limit headers, not a guess
    const wait = retryAfter(res.headers) ?? 150;
    await sleep(wait);

    if (res.status === 429) {
      await sleep(wait * 4);
      chunks.unshift(batch); // retry this batch
    }
  }
}
